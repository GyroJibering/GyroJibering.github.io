cmake_minimum_required(VERSION 3.20)
project(low_latency_lab LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(Threads REQUIRED)
enable_testing()
add_library(low_latency INTERFACE)
target_include_directories(low_latency INTERFACE include)

foreach(target stress research benchmark trading_loop)
  if(target STREQUAL "stress" OR target STREQUAL "research")
    add_executable(${target} tests/${target}.cpp)
  else()
    add_executable(${target} src/${target}.cpp)
  endif()
  target_link_libraries(${target} PRIVATE low_latency Threads::Threads)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /O2)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -O3)
  endif()
endforeach()

add_test(NAME queue_stress COMMAND stress)
add_test(NAME research_counterexamples COMMAND research)
add_test(NAME synthetic_pipeline COMMAND trading_loop)
set_tests_properties(queue_stress research_counterexamples synthetic_pipeline
  PROPERTIES TIMEOUT 60)

if(UNIX AND NOT APPLE)
  add_executable(journal_demo src/journal_demo.cpp)
  target_link_libraries(journal_demo PRIVATE low_latency Threads::Threads)
  target_compile_options(journal_demo PRIVATE -Wall -Wextra -Wpedantic -O3)
endif()
